cmake_minimum_required(VERSION 3.21)
project(NezerEmu VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
if(MSVC)
  add_compile_options(/utf-8 /W3)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
find_package(Qt6 REQUIRED COMPONENTS Widgets Network Test)
file(GLOB_RECURSE CORE_SRC CONFIGURE_DEPENDS src/*.cpp src/*.h)
list(REMOVE_ITEM CORE_SRC ${CMAKE_SOURCE_DIR}/src/main.cpp)
add_library(nezer_core STATIC ${CORE_SRC})
target_include_directories(nezer_core PUBLIC src)
set(NEZER_REPO "" CACHE STRING "GitHub owner/repo for update checks")
target_compile_definitions(nezer_core PUBLIC NEZER_VERSION="${PROJECT_VERSION}" NEZER_REPO="${NEZER_REPO}")
target_link_libraries(nezer_core PUBLIC Qt6::Widgets Qt6::Network)
if(WIN32)
  target_link_libraries(nezer_core PUBLIC user32 shell32)
  file(WRITE ${CMAKE_BINARY_DIR}/app.rc "IDI_ICON1 ICON \"${CMAKE_SOURCE_DIR}/assets/nezer.ico\"\n")
  add_executable(NezerEmu WIN32 src/main.cpp ${CMAKE_BINARY_DIR}/app.rc)
else()
  add_executable(NezerEmu src/main.cpp)
endif()
target_link_libraries(NezerEmu PRIVATE nezer_core)
add_custom_command(TARGET NezerEmu POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/assets $<TARGET_FILE_DIR:NezerEmu>/assets)
enable_testing()
foreach(t settings mousemath engine)
  add_executable(test_${t} tests/test_${t}.cpp)
  target_link_libraries(test_${t} PRIVATE nezer_core Qt6::Test)
  target_compile_definitions(test_${t} PRIVATE TEST_DATA_DIR="${CMAKE_SOURCE_DIR}/tests/data")
  add_test(NAME ${t} COMMAND test_${t})
endforeach()
